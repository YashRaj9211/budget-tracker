package database

import (
	"database/sql"
	"embed"
	"fmt"
	"log"
	"sort"
	"strings"
	"time"
)

// SchemaMigrationsTable tracks applied versioned migrations
const SchemaMigrationsTable = "schema_migrations"

type MigrationManager struct {
	db *sql.DB
	fs embed.FS
}

func NewMigrationManager(db *sql.DB, fs embed.FS) *MigrationManager {
	return &MigrationManager{db: db, fs: fs}
}

// EnsureMigrationTable ensures the schema_migrations table exists
func (m *MigrationManager) EnsureMigrationTable() error {
	query := fmt.Sprintf(`
		CREATE TABLE IF NOT EXISTS %s (
			version VARCHAR(100) PRIMARY KEY,
			applied_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
		);
	`, SchemaMigrationsTable)

	_, err := m.db.Exec(query)
	return err
}

// GetAppliedMigrations returns a map of already applied migration versions
func (m *MigrationManager) GetAppliedMigrations() (map[string]time.Time, error) {
	applied := make(map[string]time.Time)
	rows, err := m.db.Query(fmt.Sprintf("SELECT version, applied_at FROM %s ORDER BY version ASC", SchemaMigrationsTable))
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	for rows.Next() {
		var version string
		var appliedAt time.Time
		if err := rows.Scan(&version, &appliedAt); err != nil {
			return nil, err
		}
		applied[version] = appliedAt
	}
	return applied, nil
}

// RunUpMigrations applies all pending *.up.sql migrations in lexicographical order
func (m *MigrationManager) RunUpMigrations() error {
	if err := m.EnsureMigrationTable(); err != nil {
		return fmt.Errorf("failed to create migration table: %w", err)
	}

	applied, err := m.GetAppliedMigrations()
	if err != nil {
		return fmt.Errorf("failed to get applied migrations: %w", err)
	}

	entries, err := m.fs.ReadDir(".")
	if err != nil {
		return fmt.Errorf("failed to read migrations directory: %w", err)
	}

	var upFiles []string
	for _, entry := range entries {
		if !entry.IsDir() && strings.HasSuffix(entry.Name(), ".up.sql") {
			upFiles = append(upFiles, entry.Name())
		}
	}
	sort.Strings(upFiles)

	for _, filename := range upFiles {
		version := strings.TrimSuffix(filename, ".up.sql")
		if _, ok := applied[version]; ok {
			log.Printf("Migration %s already applied, skipping", version)
			continue
		}

		log.Printf("Applying migration: %s...", version)
		content, err := m.fs.ReadFile(filename)
		if err != nil {
			return fmt.Errorf("failed to read migration file %s: %w", filename, err)
		}

		tx, err := m.db.Begin()
		if err != nil {
			return fmt.Errorf("failed to start transaction: %w", err)
		}

		if _, err := tx.Exec(string(content)); err != nil {
			tx.Rollback()
			return fmt.Errorf("failed to execute migration %s: %w", filename, err)
		}

		if _, err := tx.Exec(
			fmt.Sprintf("INSERT INTO %s (version, applied_at) VALUES ($1, NOW())", SchemaMigrationsTable),
			version,
		); err != nil {
			tx.Rollback()
			return fmt.Errorf("failed to record migration %s: %w", filename, err)
		}

		if err := tx.Commit(); err != nil {
			return fmt.Errorf("failed to commit migration transaction %s: %w", filename, err)
		}

		log.Printf("Successfully applied migration: %s", version)
	}

	return nil
}

// RunDownMigration rolls back the last applied migration
func (m *MigrationManager) RunDownMigration() error {
	applied, err := m.GetAppliedMigrations()
	if err != nil {
		return fmt.Errorf("failed to get applied migrations: %w", err)
	}

	if len(applied) == 0 {
		log.Println("No migrations to rollback.")
		return nil
	}

	var versions []string
	for v := range applied {
		versions = append(versions, v)
	}
	sort.Strings(versions)
	latest := versions[len(versions)-1]

	downFilename := latest + ".down.sql"
	log.Printf("Rolling back migration: %s...", latest)

	content, err := m.fs.ReadFile(downFilename)
	if err != nil {
		return fmt.Errorf("down migration file %s not found: %w", downFilename, err)
	}

	tx, err := m.db.Begin()
	if err != nil {
		return fmt.Errorf("failed to start transaction: %w", err)
	}

	if _, err := tx.Exec(string(content)); err != nil {
		tx.Rollback()
		return fmt.Errorf("failed to execute rollback %s: %w", downFilename, err)
	}

	if _, err := tx.Exec(
		fmt.Sprintf("DELETE FROM %s WHERE version = $1", SchemaMigrationsTable),
		latest,
	); err != nil {
		tx.Rollback()
		return fmt.Errorf("failed to remove migration record %s: %w", latest, err)
	}

	if err := tx.Commit(); err != nil {
		return fmt.Errorf("failed to commit rollback transaction: %w", err)
	}

	log.Printf("Successfully rolled back migration: %s", latest)
	return nil
}
